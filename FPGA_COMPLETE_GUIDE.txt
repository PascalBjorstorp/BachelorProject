# Should show: "operating"

# Step 13: Verify addresses are accessible
sudo python3 << 'EOF'
import mmap
import os
fd = os.open("/dev/mem", os.O_RDWR | os.O_SYNC)
# DMA at 0xA0000000
dma = mmap.mmap(fd, 0x100, offset=0xA0000000)
print(f"DMA status (0xA0000000+4): {int.from_bytes(dma[4:8], 'little'):08x}")
dma.close()
# MPC at 0xA0010000
mpc = mmap.mmap(fd, 0x100, offset=0xA0010000)
print(f"MPC AP_CTRL (0xA0010000): {int.from_bytes(mpc[0:4], 'little'):08x}")
mpc.close()
os.close(fd)
print("Addresses accessible!")
EOF

# Expected output:
# DMA status: 0x00000001 (halted - OK)
# MPC AP_CTRL: 0x00000004 (idle - OK)

# =============================================
# PART 7: SD CARD CORRUPTION RECOVERY + PREVENTION
# =============================================

# Typical symptoms after premature shutdown:
# - Boot hangs or drops to emergency mode
# - Filesystem becomes read-only
# - Missing/corrupted files in /boot or rootfs
# - FAT "dirty bit" warnings

# Step 17: Power off Ultra96, remove SD card, insert into Linux host
# IMPORTANT: Never run fsck on mounted partitions.

# Step 18: Confirm device is /dev/sda in your fixed setup
lsblk -o NAME,SIZE,FSTYPE,MOUNTPOINT,LABEL

# This guide assumes your fixed setup:
#   /dev/sda1 = boot (vfat)
#   /dev/sda2 = rootfs (ext4)
#   /dev/sda  = full SD device

# Step 19: Unmount all SD partitions
sudo umount /dev/sda1 2>/dev/null || true
sudo umount /dev/sda2 2>/dev/null || true

# Optional dry-run checks (read-only, no modifications)
sudo fsck.vfat -vn /dev/sda1
sudo e2fsck -fn /dev/sda2

# Step 20: Repair boot partition (FAT)
sudo fsck.vfat -a /dev/sda1

# Step 21: Repair root filesystem (ext4)
sudo e2fsck -f -y /dev/sda2

# Step 22: Re-check that both partitions are clean
sudo fsck.vfat -vn /dev/sda1
sudo e2fsck -fn /dev/sda2
# Expected: no remaining filesystem errors.

# Optional (if corruption repeats): check media health (slow)
sudo badblocks -sv /dev/sda

# Step 23: Mount and verify key files still exist
sudo mkdir -p /mnt/sdboot /mnt/sdroot
sudo mount /dev/sda1 /mnt/sdboot
sudo mount /dev/sda2 /mnt/sdroot
ls /mnt/sdboot
ls /mnt/sdroot/home/xilinx
sudo umount /mnt/sdboot
sudo umount /mnt/sdroot

# Step 24: Reinsert SD card and boot Ultra96
# If boot still fails repeatedly, reflash the SD image and recopy project files.

# PREVENTION: Always clean shutdown before removing power
sync
sudo poweroff
# Wait until activity LEDs stop before unplugging power.
